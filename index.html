#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

#ifdef _WIN32
#include <string.h>
#define COMPARA_STRING _stricmp
#else
#include <strings.h>
#define COMPARA_STRING strcasecmp
#endif

typedef struct Node {
    char *nome_tag;
    int linha;
    struct Node *proximo;
} Node;

typedef struct Pilha {
    Node *topo;
    int tamanho;
} Pilha;

Pilha* criar(void) {
    Pilha *p = (Pilha*) malloc(sizeof(Pilha));
    if (p != NULL) {
        p->topo = NULL;
        p->tamanho = 0;
    }
    return p;
}

bool vazia(Pilha *p) {
    if (p == NULL) return true;
    return p->topo == NULL;
}

int tamamho(Pilha *p) {
    if (p == NULL) return 0;
    return p->tamanho;
}

bool empilhar(Pilha *p, const char *nome_tag, int linha) {
    if (p == NULL || nome_tag == NULL) return false;

    Node *novo = (Node*) malloc(sizeof(Node));
    if (novo == NULL) return false;

    novo->nome_tag = (char*) malloc(strlen(nome_tag) + 1);
    if (novo->nome_tag == NULL) {
        free(novo);
        return false;
    }
    strcpy(novo->nome_tag, nome_tag);

    novo->linha = linha;
    novo->proximo = p->topo;
    p->topo = novo;
    p->tamanho++;

    return true;
}

bool desempilhar(Pilha *p, char **nome_tag, int *linha) {
    if (p == NULL || vazia(p)) return false;

    Node *remover = p->topo;

    if (nome_tag != NULL) {
        *nome_tag = remover->nome_tag; 
    } else {
        free(remover->nome_tag);
    }

    if (linha != NULL) {
        *linha = remover->linha;
    }

    p->topo = remover->proximo;
    free(remover);
    p->tamanho--;

    return true;
}

void destruir(Pilha *p) {
    if (p == NULL) return;

    while (!vazia(p)) {
        desempilhar(p, NULL, NULL);
    }
    free(p);
}

bool eh_tag_orfa(const char *tag) {
    const char *orfas[] = {"br", "hr", "img", "input", "meta", "link"};
    int total_orfas = sizeof(orfas) / sizeof(orfas[0]);

    for (int i = 0; i < total_orfas; i++) {
        if (COMPARA_STRING(tag, orfas[i]) == 0) {
            return true;
        }
    }
    return false;
}

int main(int argc, char *argv[]) {
    if (argc < 2) {
        printf("Uso: %s <arquivo.html>\n", argv[0]);
        return 1;
    }

    FILE *arquivo = fopen(argv[1], "r");
    if (arquivo == NULL) {
        printf("Erro ao abrir o arquivo %s\n", argv[1]);
        return 1;
    }

    Pilha *pilha = criar();
    if (pilha == NULL) {
        fclose(arquivo);
        return 1;
    }

    int linha_atual = 1;
    int ch;

    while ((ch = fgetc(arquivo)) != EOF) {
       if (ch == '\n') {
           linha_atual++;
           continue;
         }

        if (ch == '<') {
            int proximo = fgetc(arquivo);

            if (proximo == '!') {
                while ((ch = fgetc(arquivo)) != EOF && ch != '>') {
                  if (ch == '\n') linha_atual++;
                }
                 continue;
           }

            bool eh_fechamento = false;
            if (proximo == '/') {
                eh_fechamento = true;
            } else {
                ungetc(proximo, arquivo); 
            }

            char buffer_tag[256];
            int idx = 0;

            while ((ch = fgetc(arquivo)) != EOF && ch != '>' && ch != ' ' && ch != '\t' && ch != '\n' && ch != '\r') {
                if (idx < (int)sizeof(buffer_tag) - 1) {
                    buffer_tag[idx++] = (char)ch;
                }
            }
            
            if (ch == '\n') {
                linha_atual++;
            }
            
            buffer_tag[idx] = '\0';

            while (ch != '>' && ch != EOF) {
                if (ch == '\n') linha_atual++;
                ch = fgetc(arquivo);
            }

            if (idx == 0) continue; 

            if (eh_fechamento) {
                if (vazia(pilha)) {
                    printf("Mal formatado: Tag de fechamento </%s> inesperada na linha %d\n", buffer_tag, linha_atual);
                    destruir(pilha);
                    fclose(arquivo);
                    return 0;
                }

                char *tag_topo = NULL;
                int linha_topo = 0;
                desempilhar(pilha, &tag_topo, &linha_topo);

                if (COMPARA_STRING(buffer_tag, tag_topo) != 0) {
                    printf("Mal formatado: Tag de fechamento </%s> inesperada na linha %d\n", buffer_tag, linha_atual);
                    free(tag_topo);
                    destruir(pilha);
                    fclose(arquivo);
                    return 0;
                }

                free(tag_topo);
            } 
            else {
                if (!eh_tag_orfa(buffer_tag)) {
                    empilhar(pilha, buffer_tag, linha_atual);
                }
            }
        }
    }

    if (!vazia(pilha)) {
        char *tag_topo = NULL;
        int linha_topo = 0;
        desempilhar(pilha, &tag_topo, &linha_topo);

        printf("Mal formatado: Tag <%s> aberta na linha %d nao foi fechada\n", tag_topo, linha_topo);
        free(tag_topo);
        destruir(pilha);
        fclose(arquivo);
        return 0;
    }

    printf("Bem formatado\n");

    destruir(pilha);
    fclose(arquivo);
    return 0;
}
